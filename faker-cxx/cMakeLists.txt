cmake_minimum_required(VERSION 3.14)
project(MeuProjetoFaker CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

set(FAKER_BUILD_TESTING OFF)

FetchContent_Declare(
    faker-cxx
    GIT_REPOSITORY https://github.com/cieslarmichal/faker-cxx.git
    GIT_TAG main
)

FetchContent_MakeAvailable(faker-cxx)

add_executable(myapp testeFaker.cpp)

target_link_libraries(myapp PRIVATE faker-cxx)
